def create_metadata_dict(file_path):
    MetaDataValueJson = {}

    file_extension = file_path.split('.')[-1]
    
    if file_extension == 'xlsx':
        excel_file_df = pd.ExcelFile(file_path, engine="openpyxl")
        for sheet_name in excel_file_df.sheet_names:
            excel_sheet_df = pd.read_excel(file_path, dtype=str, sheet_name=sheet_name, engine="openpyxl")
            #excel_sheet_df.columns = (excel_sheet_df.columns.str.strip().str.lower())
            MetaDataValueJson[sheet_name] = excel_sheet_df
    else:
        raise Exception(f"File {file_path} is not an Excel File")

    return MetaDataValueJson

def merge_dataframe_to_postgres(
    df: pd.DataFrame,
    engine,
    merge_sql_template: str,
    table_name: str,
    table_schema: str,
    temp_schema: str = "staging",
    temp_table_prefix: str = "temp"
):
    temp_table_name = f"{temp_table_prefix}_{table_name}".lower()
    return_dict = {'TableName': table_name, 'TempTableName': temp_table_name, 'Status': 'Pass','Message': '', 'DFLength': 0,'TempTableLength': 0, 'TableLength': 0}
    try:
        df.columns = (df.columns.str.strip().str.lower())
        length_of_df = len(df)
        df.to_sql(name=temp_table_name, con=engine, schema=temp_schema, if_exists="replace", index=False, method='multi', chunksize=5000)

        # Verify insert by querying temp table row count
        with engine.connect() as verify_conn:
            temp_count = verify_conn.execute(text(f"SELECT COUNT(*) FROM {temp_schema}.{temp_table_name}")).fetchone()[0]

        return_dict['DFLength'] = length_of_df
        return_dict['TempTableLength'] = temp_count

        with engine.begin() as conn:
            # Get Merge
            merge_sql = merge_sql_template.format(temp_schema=temp_schema,temp_table_name=temp_table_name,table_schema=table_schema)
            
            # Run Merge
            conn.execute(text(merge_sql))

            # Drop Temp Table
            conn.execute(text(f"DROP TABLE IF EXISTS {temp_schema}.{temp_table_name} ; "))

            # Get Table Length
            table_length_sql = f"SELECT COUNT(*) FROM {table_schema}.{table_name}"
            table_length = conn.execute(text(table_length_sql)).fetchone()[0]
            return_dict['TableLength'] = table_length

        return return_dict
    except Exception as e:
        return_dict['Status'] = 'Fail'
        return_dict['Message'] = str(e)
        return return_dict
